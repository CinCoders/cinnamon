export const InstallCommand = ({ command }) => (
  <div className="about-install-command" onClick={(event) => event.stopPropagation()}>
    <code>{command}</code>
    <button
      type="button"
      className="about-copy-button"
      onClick={async (event) => {
        event.preventDefault();
        event.stopPropagation();

        const button = event.currentTarget;
        const originalLabel = button.textContent;

        try {
          await navigator.clipboard.writeText(command);
          button.textContent = "Copied";
        } catch {
          button.textContent = "Copy failed";
        }

        window.setTimeout(() => {
          button.textContent = originalLabel;
        }, 1500);
      }}
    >
      Copy
    </button>

  </div>
);

# Welcome to Cinnamon's Storybook

Cinnamon is a powerful React package that offers a variety of components designed to simplify the development process and enhance the appearance of systems across multiple projects.
With its reusable components, Cinnamon helps standardize authenticated application shells across different projects and teams.
In the v2 direction, the library replaces the legacy MUI and styled-components stack with a lighter Tailwind-based foundation and Radix/Shadcn-style primitives.
Whether you're working on a React SPA or a server-oriented setup, Cinnamon aims to streamline development and keep layout, navigation, and auth-related composition consistent.
