import { redirect } from "next/navigation";
import { PageWithAuthServer } from "@cincoders/cinnamon/server";

export default async function ProtectedAppPage() {
  const session = { isAuthenticated: true, roles: ["admin"] };

  return (
    <PageWithAuthServer
      authProps={{
        session,
        permittedRoles: ["admin"],
        onUnauthenticated: () => redirect("http://localhost:8080/"),
      }}
      centralized={false}
      flexDirection="column"
    >
      <div>Conteúdo protegido renderizando no SERVER ✅</div>
    </PageWithAuthServer>
  );
}
